2. Reach the frontend:

{{- if .Values.ingress.enabled }}
     Via Ingress:
{{- range .Values.ingress.hosts }}
       http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}
{{- end }}
{{- else if eq .Values.frontend.service.type "LoadBalancer" }}
     Via LoadBalancer (watch for the external IP to get assigned):

       kubectl -n {{ .Release.Namespace }} get svc {{ include "spatiumddi.fullname" . }}-frontend -w

{{- else }}
     No Ingress and ClusterIP service — port-forward for local access:

       kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "spatiumddi.fullname" . }}-frontend 8080:{{ .Values.frontend.service.port }}
       open http://localhost:8080
{{- end }}
